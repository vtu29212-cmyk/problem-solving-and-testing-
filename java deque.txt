import java.util.*;

public class Solution {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        int[] nums = new int[n];

        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }

        Deque<Integer> deque = new ArrayDeque<>();
        Set<Integer> set = new HashSet<>();

        int maxUnique = 0;

        for (int i = 0; i < n; i++) {

            // Add new element to the window
            deque.addLast(nums[i]);
            set.add(nums[i]);

            // If window size becomes greater than m
            if (deque.size() > m) {
                int removed = deque.removeFirst();

                // Remove from set only if it doesn't
                // exist elsewhere in the current window
                if (!deque.contains(removed)) {
                    set.remove(removed);
                }
            }

            // Check maximum number of unique elements
            if (deque.size() == m) {
                maxUnique = Math.max(maxUnique, set.size());
            }
        }

        System.out.println(maxUnique);

        sc.close();
    }
}
